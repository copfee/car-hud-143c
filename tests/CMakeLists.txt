cmake_minimum_required(VERSION 3.16)
project(hud_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
execute_process(COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/../tools/generate_profiles.py" COMMAND_ERROR_IS_FATAL ANY)
get_filename_component(HUD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${HUD_ROOT}/src/hud/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${HUD_ROOT}/.pio/libdeps/waveshare_amoled_143c/lvgl" "lvgl")
add_executable(core_test core_test.cpp "${HUD_ROOT}/src/hud/core.cpp" "${HUD_ROOT}/src/hud/obd.cpp")
target_include_directories(core_test PRIVATE "${HUD_ROOT}/src/hud")
add_executable(ui_test ui_test.cpp "${HUD_ROOT}/src/hud/core.cpp" "${HUD_ROOT}/src/hud/obd.cpp" "${HUD_ROOT}/src/hud/ui.cpp" "${HUD_ROOT}/src/hud/hud_cn_18.c" "${HUD_ROOT}/src/hud/hud_speed_96.c" "${HUD_ROOT}/src/hud/mercedes_logo.c")
target_include_directories(ui_test PRIVATE "${HUD_ROOT}/src/hud")
target_compile_definitions(ui_test PRIVATE LV_LVGL_H_INCLUDE_SIMPLE)
target_link_libraries(ui_test PRIVATE lvgl)
enable_testing()
add_test(NAME protocol_and_model COMMAND core_test)
add_test(NAME rendered_ui COMMAND ui_test "${CMAKE_BINARY_DIR}")
add_test(NAME rendered_ui_rgb565 COMMAND ui_test "${CMAKE_BINARY_DIR}/rgb565" --rgb565)
function(add_swipe_bench name ui_source)
    add_executable(${name} swipe_bench.cpp "${HUD_ROOT}/src/hud/core.cpp" "${HUD_ROOT}/src/hud/obd.cpp" "${ui_source}" "${HUD_ROOT}/src/hud/hud_cn_18.c" "${HUD_ROOT}/src/hud/hud_speed_96.c" "${HUD_ROOT}/src/hud/mercedes_logo.c")
    target_include_directories(${name} PRIVATE "${HUD_ROOT}/src/hud")
    target_compile_definitions(${name} PRIVATE LV_LVGL_H_INCLUDE_SIMPLE)
    target_link_libraries(${name} PRIVATE lvgl)
endfunction()
add_swipe_bench(swipe_bench "${HUD_ROOT}/src/hud/ui.cpp")
if(HUD_BASELINE_UI)
    add_swipe_bench(swipe_bench_before "${HUD_BASELINE_UI}")
endif()
add_executable(obd_test obd_test.cpp "${HUD_ROOT}/src/hud/core.cpp" "${HUD_ROOT}/src/hud/obd.cpp")
target_include_directories(obd_test PRIVATE "${HUD_ROOT}/src/hud")
add_test(NAME universal_obd COMMAND obd_test "${HUD_ROOT}/tests/fixtures/protocol.tsv")
